cmake_minimum_required(VERSION 3.20)
project(SRN_Client LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(SRN_Client SHARED src/dllmain.cpp src/srn_client.cpp)
target_include_directories(SRN_Client PRIVATE include)
target_compile_definitions(SRN_Client PRIVATE UNICODE _UNICODE SRN_CLIENT_EXPORTS)
if(MSVC)
  target_compile_options(SRN_Client PRIVATE /W4 /O2)
endif()
